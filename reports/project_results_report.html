Project Report: Software Vulnerability Classification (CWE)

Objective: To develop a highly efficient, state-of-the-art machine learning pipeline for detecting and classifying Common Weakness Enumeration (CWE) vulnerabilities in source code using lightweight, Transformer-based linguistic models.

1. Sample Code Snippets

Preprocessing Code

import re

def clean_code(code):
    if not isinstance(code, str): return ""
    
    # Remove single-line and multi-line comments
    code = re.sub(r"//.*", "", code)
    code = re.sub(r"/\*.*?\*/", "", code, flags=re.DOTALL)
    
    # Normalize whitespace
    code = re.sub(r"\s+", " ", code)
    return code.strip()

Tokenizer Usage

from transformers import AutoTokenizer

tokenizer = AutoTokenizer.from_pretrained("Salesforce/codet5-small")

Model Training Snippet

def train_one_epoch(model, dataloader, optimizer, device):
    model.train()
    # Using Class-Weighted Cross Entropy to handle dataset imbalance
    loss_fn = nn.CrossEntropyLoss(weight=class_weights)
    
    for batch in dataloader:
        outputs = model(
            input_ids=batch["input_ids"].to(device), 
            attention_mask=batch["attention_mask"].to(device)
        )
        loss = loss_fn(outputs.logits, batch["labels"].to(device))
        
        loss.backward()
        optimizer.step()
        optimizer.zero_grad()
